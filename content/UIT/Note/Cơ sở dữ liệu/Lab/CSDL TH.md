CREATE TABLE QLKH.SACH (
    MaSach CHAR(5) NOT NULL,
    TenSach NVARCHAR(20),
    TheLoai NVARCHAR(50),
    NXB NVARCHAR(20),
    CONSTRAINT PK_SACH 
    	PRIMARY KEY (MaSach)
);

CREATE TABLE QLKH.KHACHHANG (
    MaKH CHAR(5) NOT NULL,
    HoTen NVARCHAR(25),
    NgaySinh SMALLDATETIME,
    DiaChi NVARCHAR(50),
    SoDT VARCHAR(15),
    NgDK SMALLDATETIME,
    CONSTRAINT PK_KHACHHANG 
    	PRIMARY KEY (MaKH)
);

CREATE TABLE QLKH.PHIEUMUA (
    MaPM CHAR(5) NOT NULL,
    MaKH CHAR(5),
    NgayMua SMALLDATETIME,
    SoSachMua INT,
    CONSTRAINT PK_PHIEUMUA 
    	PRIMARY KEY (MaPM),
    CONSTRAINT FK_PHIEUMUA_KHACHHANG
        FOREIGN KEY (MaKH) REFERENCES QLKH.KHACHHANG(MaKH)
);

CREATE TABLE QLKH.CHITIET_PHIEUMUA (
    MaSach CHAR(5) NOT NULL,
    MaPM CHAR(5) NOT NULL,
    CONSTRAINT PK_CHITIET_PHIEUMUA 
    	PRIMARY KEY (MaSach, MaPM),
    CONSTRAINT FK_CHITIET_SACH
        FOREIGN KEY (MaSach) REFERENCES QLKH.SACH(MaSach),
    CONSTRAINT FK_CHITIET_PHIEUMUA
        FOREIGN KEY (MaPM) REFERENCES QLKH.PHIEUMUA(MaPM)
);
